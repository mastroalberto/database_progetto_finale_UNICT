    \textbf{Termine} & \textbf{Descrizione} & \textbf{Sinonimo} & \textbf{Legame} \\
    \hline
    \endhead 
    \textbf{Docente} & Professionista incaricato di svolgere le lezioni & Insegnante & Aula, Lezione \\
    \hline
    \textbf{Aula} & Luogo dove si svolgono le lezioni & Classe & Aula, Lezione \\
    \hline
    \textbf{Strumento} & Attrezzatura musicale che può essere fissa o mobile e noleggiata & - & Noleggio, Lezione \\
    \hline
    \textbf{Studente} & Persona che segue le lezioni del docente & Allievo & Noleggio, Lezione, Aula, Studente \\
    \hline
    \textbf{Lezione} & Ora di insegnamento dello strumento & Insegnamento & Aula, Lezione, Presenze, Pagamento \\
    \hline
    \textbf{Esame} & Esame di livello svolto dallo studente & Prova & Studente \\
    \hline
    \textbf{Noleggio} & Contratto di locazione di uno strumento a tempo & Affitto & Studente, Magazzino \\
    \hline
    \textbf{Prodotto} & Oggetto da acquistare & Oggetto, Accessorio & Magazzino \\ 
    \hline
    \textbf{Concerto} & Attività musicale e dimostrativa & Esibizione, Spettacolo & Studente, Docente \\
    \hline
    \textbf{Workshop} & Lezione frontale di gruppo con docenti esterni & Laboratorio, Seminario & Aula, Lezione \\
